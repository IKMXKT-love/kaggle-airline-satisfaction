# Experiments
- **Expt 1:** Single CPU - LGBM Baseline model trained of Competition dataset. **LB -> 0.95905** 
- **Expt 2:** Feature engineering of the Competition dataset and reduction of features having signification up to 0.70 (`FEATURE_KEEP_RATIO = 0.70`). **LB -> 0.95765**
- **Expt 3:** Same Feature engineering, Slightly increased Feature Significance (`FEATURE_KEEP_RATIO = 0.75`). **LB -> 0.95787**
- **Expt 4:** Removed Feature engineering as it is not useful, but, created a new features by considering Flight Distance as a route ID, target encoding, value counts, route profile as suggested by [goodpjw2008](https://www.kaggle.com/code/goodpjw2008/s6e10-auxiliary-task-features-lb-0-96129). **LB -> 0.96025**
- **Expt 5:** Kept a Common Random Seed (`SEED = 42`), Changed `N_FOLDS` from 5 to 10 (`N_FOLDS = 10`) and LBGM model `learning_rate": 0.01`. No much improvement in LB score with doubling of `N_FOLDS` and consume a lot of CPU runtime. **LB -> 0.96027**
- **Expt 6:** Apply logit transform, Revert to `N_FOLDS = 5`. The mean submission file LB reduced by -0.00009 with **LB -> 0.96016**. The private roc-auc score for logit transform was **0.98659** high but not peformed well in **LB -> 0.95706**. The overfitting is due to huge leakage.
- **Expt 7:** Blending of prediction probability by CV, Full data and logit with various weights. **LB -> 0.96014**
- **Expt 8:** Nested Log.Reg. in logit space. No improvement in LB, but has better CV ROC-AUC score. 🤔 😠 **LB -> 0.95706**
- **Expt 9:** Training the Original dataset. The CV score is high as 0.99480. But LB plunged to **LB -> 0.95429**
- **Expt 10** Training with Original dataset decays the LB score. So, removing Original dataset training and sticking to Competetion Dataset.

def audit_missing_and_negative(df, name="df"):
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    report = []
    total_missing = 0
    total_negative = 0
    for c in df.columns:
        n_miss = int(df[c].isna().sum())
        n_neg  = 0
        if c in numeric_cols:
            n_neg = int((df[c] < 0).sum())
        total_missing += n_miss
        total_negative += n_neg
        if n_miss > 0 or n_neg > 0:
            report.append({"column": c, "dtype": str(df[c].dtype),
                           "missing": n_miss, "negative": n_neg})
    rep_df = pd.DataFrame(report)
    print(f"\n{'='*70}\nMISSING / NEGATIVE REPORT — {name}\n{'='*70}")
    if len(rep_df) == 0:
        print("No missing or negative values found.")
    else:
        print(rep_df.to_string(index=False))
    print(f"\nTotal missing values : {total_missing}")
    print(f"Total negative values: {total_negative}")
    return total_missing, total_negative

def impute_mean(df, train_df, cols):
    df = df.copy()
    impute_report = []
    for c in cols:
        if c not in df.columns:
            continue
        train_mean = train_df[c].mean()
        mask_miss = df[c].isna()
        n_miss = int(mask_miss.sum())
        if n_miss > 0:
            df.loc[mask_miss, c] = train_mean
        mask_neg = df[c] < 0
        n_neg = int(mask_neg.sum())
        if n_neg > 0:
            df.loc[mask_neg, c] = train_mean
        if n_miss > 0 or n_neg > 0:
            impute_report.append((c, n_miss, n_neg, train_mean))
    return df, impute_report

def add_route_features(train_df, test_df, service_cols):
    # ---- Route ID ----
    train_df["Route_ID"] = train_df["Flight Distance"].astype(int)
    test_df["Route_ID"]  = test_df["Flight Distance"].astype(int)

    # ---- Route popularity ----
    all_routes = pd.concat(
        [train_df["Route_ID"], test_df["Route_ID"]],
        ignore_index=True)

    route_counts = all_routes.value_counts()
    total_flights = len(all_routes)

    for df in [train_df, test_df]:
        df["Route_Freq"] = (
            df["Route_ID"].map(route_counts).astype(np.int32))
        df["Route_Freq_Ratio"] = df["Route_Freq"] / total_flights

    freq_q75 = route_counts.quantile(0.75)

    for df in [train_df, test_df]:
        df["Route_Is_Common"] = (df["Route_Freq"] >= freq_q75).astype(np.int8)

    # ---- Route profile (train only) ----
    route_profile_cols = (
        ["Age", "Departure Delay in Minutes","Arrival Delay in Minutes"]
        + service_cols)

    route_profile = (
        train_df.groupby("Route_ID")[route_profile_cols]
        .mean().rename(
            columns=lambda c:
            f"Route_Mean_{c.replace(' ', '_').replace('/', '_')}"
        ).reset_index()
    )

    train_df = train_df.merge(
        route_profile, on="Route_ID", how="left")

    test_df = test_df.merge(
        route_profile, on="Route_ID", how="left")

    # ---- Statistics ----
    print("\nRoute statistics:")
    print(f"  unique routes    : {route_counts.shape[0]}")
    print(f"  max frequency    : {route_counts.max()}")
    print(f"  min frequency    : {route_counts.min()}")
    print(f"  median frequency : {route_counts.median():.0f}")
    print(f"  75th percentile  : {freq_q75:.0f}")

    return train_df, test_df, route_profile

# ---- 3d. Target encoding (K-fold, smoothed) — no leakage into train ----
def target_encode_kfold(train_df, test_df, col, target,
                        n_folds=5, smoothing=20.0, seed=SEED):
    """Out-of-fold smoothed target-mean encoding."""
    train_encoded = np.full(len(train_df), np.nan, dtype=np.float32)
    global_mean = train_df[target].astype(float).mean()

    kf = KFold(n_splits=n_folds, shuffle=True, random_state=seed)
    for tr_idx, va_idx in kf.split(train_df):
        tr_data = train_df.iloc[tr_idx]
        agg = tr_data.groupby(col)[target].agg(["mean", "count"])
        smoothed = (agg["mean"] * agg["count"] + global_mean * smoothing) / (agg["count"] + smoothing)
        mapping = smoothed.to_dict()
        train_encoded[va_idx] = (
            train_df.iloc[va_idx][col].map(mapping).fillna(global_mean).values
        )

    agg = train_df.groupby(col)[target].agg(["mean", "count"])
    smoothed = (agg["mean"] * agg["count"] + global_mean * smoothing) / (agg["count"] + smoothing)
    mapping = smoothed.to_dict()
    test_encoded = test_df[col].map(mapping).fillna(global_mean).values.astype(np.float32)
    return train_encoded, test_encoded


def target_std_kfold(train_df, test_df, col, target,
                     n_folds=5, smoothing=20.0, seed=SEED):
    """Out-of-fold smoothed target-std per route (volatility)."""
    train_std = np.full(len(train_df), np.nan, dtype=np.float32)
    global_std = train_df[target].astype(float).std()

    kf = KFold(n_splits=n_folds, shuffle=True, random_state=seed)
    for tr_idx, va_idx in kf.split(train_df):
        tr_data = train_df.iloc[tr_idx]
        agg = tr_data.groupby(col)[target].agg(["std", "count"])
        agg["std"] = agg["std"].fillna(global_std)
        smoothed = (agg["std"] * agg["count"] + global_std * smoothing) / (agg["count"] + smoothing)
        mapping = smoothed.to_dict()
        train_std[va_idx] = (
            train_df.iloc[va_idx][col].map(mapping).fillna(global_std).values
        )

    agg = train_df.groupby(col)[target].agg(["std", "count"])
    agg["std"] = agg["std"].fillna(global_std)
    smoothed = (agg["std"] * agg["count"] + global_std * smoothing) / (agg["count"] + smoothing)
    mapping = smoothed.to_dict()
    test_std = test_df[col].map(mapping).fillna(global_std).values.astype(np.float32)
    return train_std, test_std

def run_cv(X_data, y_data, X_test_data, params, n_folds=N_FOLDS,
           tag="", return_feat_imp=False):
    skf = StratifiedKFold(n_splits=n_folds, shuffle=True, random_state=SEED)
    oof = np.zeros(len(X_data))
    test_pred = np.zeros(len(X_test_data))
    fold_aucs = []
    best_iters = []
    feat_imp_gain = np.zeros(X_data.shape[1]) if return_feat_imp else None

    for fold, (tr_idx, va_idx) in enumerate(skf.split(X_data, y_data), 1):
        X_tr, X_va = X_data.iloc[tr_idx], X_data.iloc[va_idx]
        y_tr, y_va = y_data[tr_idx], y_data[va_idx]

        dtr = lgb.Dataset(X_tr, y_tr)
        dva = lgb.Dataset(X_va, y_va, reference=dtr)

        gbm = lgb.train(
            params, dtr, num_boost_round=10000,
            valid_sets=[dva],
            callbacks=[lgb.early_stopping(200, verbose=False)],
        )

        va_pred = gbm.predict(X_va, num_iteration=gbm.best_iteration)
        oof[va_idx] = va_pred
        fold_aucs.append(roc_auc_score(y_va, va_pred))
        best_iters.append(gbm.best_iteration)
        test_pred += gbm.predict(X_test_data, num_iteration=gbm.best_iteration) / n_folds

        if return_feat_imp:
            feat_imp_gain += gbm.feature_importance(importance_type="gain")

        print(f"  [{tag}] Fold {fold:2d}/{n_folds}  AUC={fold_aucs[-1]:.5f}  "
              f"best_iter={gbm.best_iteration}")

    oof_auc = roc_auc_score(y_data, oof)
    print(f"\n[{tag}] OOF AUC        : {oof_auc:.5f}")
    print(f"[{tag}] Mean fold AUC  : {np.mean(fold_aucs):.5f} (+/- {np.std(fold_aucs):.5f})")
    print(f"[{tag}] Mean best_iter : {int(np.mean(best_iters))}")

    result = {"oof": oof, "test": test_pred, "oof_auc": oof_auc,
              "fold_aucs": fold_aucs, "best_iters": best_iters}
    if return_feat_imp:
        result["feat_imp"] = feat_imp_gain
    return result, gbm
