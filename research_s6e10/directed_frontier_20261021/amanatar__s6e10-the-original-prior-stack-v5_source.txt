def std_cols(df):
    ren = {}
    for c in df.columns:
        key = canon_key(c)
        if key in NAMES:
            ren[c] = NAMES[key]
        elif key.startswith('unnamed'):
            ren[c] = 'DROPME'
    out = df.rename(columns=ren)
    return out.loc[:, [c for c in out.columns if c != 'DROPME']]

def load_labels(s):
    if s.dtype == bool:
        return s.astype('int8')

              'true': 1, 'false': 0, '1': 1, '0': 0, 'y': 1, 'n': 0}
        return s.astype(str).str.strip().str.lower().map(mp).astype('int8')
    return pd.to_numeric(s, errors='raise').astype('int8')

# ---- base FE block: exact parity with our champion (EDA notebook sec 15) ----
def add_base_fe(df):
    df = df.copy()
    df['wifi_na']  = (df['Wifi'] == 0).astype('int8')
    df['eob_zero'] = (df['Ease of Booking'] == 0).astype('int8')
    df['ob_zero']  = (df['Online Boarding'] == 0).astype('int8')
    df['n_zero']   = (df[RATING] == 0).sum(axis=1).astype('int16')
    for c in ('Dep Delay', 'Arr Delay'):
        df[c + ' log'] = np.log1p(df[c].fillna(0))
    df['arr_nan']   = df['Arr Delay'].isna().astype('int8')
    df['Arr Delay'] = df['Arr Delay'].fillna(0)
    df['dep_zero']  = (df['Dep Delay'] == 0).astype('int8')
    arr = df[RATING].to_numpy(dtype=np.float64)
    ok = arr > 0
    svc = np.where(ok, arr, np.nan)
    with np.errstate(invalid='ignore'):
        sm = np.nanmean(svc, axis=1)
    fill = float(np.nanmean(sm)) if np.isfinite(sm).any() else 3.0
    df['svc_mean'] = np.nan_to_num(sm, nan=fill)
    vc = np.stack([((arr == v) & ok).sum(axis=1) for v in range(1, 6)], axis=1).astype(np.float64)
    tot = vc.sum(axis=1, keepdims=True)
    p = vc / np.where(tot > 0, tot, 1.0)
    with np.errstate(divide='ignore', invalid='ignore'):
        H = -np.nansum(np.where(p > 0, p * np.log(p), 0.0), axis=1)
    df['svc_entropy'] = np.nan_to_num(H / np.log(5), nan=0.0)
    df['Wifi'] = df['Wifi'].replace(0, np.nan)   # sentinel -> NaN for LGBM (champion parity)
    return df

# v4: triple smoothing (auto / 10 / 100) via sklearn's cross-fitting TargetEncoder -
# evgendvorkin's exact ladder, replacing v3's single m=100 manual encoder.
# fit_transform cross-fits INSIDE the fold-train (no self-leak); transform encodes
# val/test with the full fold-train mapping. keys x 3 smoothings = 3x TE columns.
from sklearn.preprocessing import TargetEncoder
TE_SMOOTHS = ['auto', 10.0, 100.0]

def te3_fit_apply(fit_fr, fit_y, parts):
    out = {name: [] for name, _ in parts}
    for sm in TE_SMOOTHS:
        enc = TargetEncoder(smooth=sm, cv=5, shuffle=True, random_state=SEED)
        Zf = enc.fit_transform(fit_fr[TE_COLS].astype(str), np.asarray(fit_y, dtype=float))
        for name, fr in parts:
            if fr is fit_fr:
                out[name].append(Zf.astype(np.float32))
            else:
                out[name].append(enc.transform(fr[TE_COLS].astype(str)).astype(np.float32))
    return {name: np.hstack(blocks) for name, blocks in out.items()}

def profile_fit(fit_fr):
    g = fit_fr.groupby('Flight Distance')[PROFILE_SRC].agg(['mean', 'std'])
    return g, fit_fr[PROFILE_SRC].mean(), fit_fr[PROFILE_SRC].std()

def profile_apply(fr, grp, defm, defs):
    fd = fr['Flight Distance']
    K = len(PROFILE_SRC)
    out = np.empty((len(fr), 3 * K), dtype=np.float32)
    for j, c in enumerate(PROFILE_SRC):
        v = np.nan_to_num(fr[c].to_numpy(dtype=np.float64), nan=0.0)
        pm = fd.map(grp[(c, 'mean')]).fillna(defm[c]).to_numpy()
        ps = fd.map(grp[(c, 'std')]).fillna(defs[c]).to_numpy()
        out[:, j] = pm
        out[:, K + j] = np.nan_to_num(ps, nan=0.0)
        out[:, 2 * K + j] = v - pm
    return out

def cont_block(fr, cols=None):
    cols = CONT_FE if cols is None else cols
    X = fr[cols].to_numpy(dtype=np.float64, copy=True)
    return np.nan_to_num(X, nan=0.0, posinf=0.0, neginf=0.0).astype(np.float32)

def build_dyn(fit_fr, fit_y, parts):
    'dynamic continuous block: global cont + triple TE + route profiles (mean/std/resid)'
    te3 = te3_fit_apply(fit_fr, fit_y, parts)
    grp, defm, defs = profile_fit(fit_fr)
    out = {}
    for name, fr in parts:
        out[name] = np.hstack([cont_block(fr), te3[name],
                               profile_apply(fr, grp, defm, defs)]).astype(np.float32)
    return out

# robust scale + smooth saturation (RealMLP-flavoured): z/(1+|z|/8) is linear near 0,
# saturates around +-8, kills the delay tails without a hard cliff
def fit_scale(A):
    med = np.median(A, axis=0)
    q1, q3 = np.percentile(A, 25, axis=0), np.percentile(A, 75, axis=0)
    iqr = np.where(q3 - q1 < 1e-9, 1.0, q3 - q1)
    return med.astype(np.float32), iqr.astype(np.float32)

def apply_scale(A, med, iqr):
    z = (A - med) / iqr
    return (z / (1.0 + np.abs(z) / 8.0)).astype(np.float32)
