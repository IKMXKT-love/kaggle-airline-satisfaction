## 3. Evidence-Based Feature Engineering

Our feature engineering pipeline incorporates 4 empirically verified pillars:
1. **In-Flight Survey Aggregates**: Row-wise statistics (mean, std, min, max, high-satisfaction count $\ge 4$, zero-count).
2. **Flight Delay Dynamics**: Total delay, departure-arrival difference (in-air recovery), log-delays, and delay-per-mile.
3. **Route-Profile Fingerprints**: Exact `Flight Distance` is used as a proxy for physical flight routes. We aggregate rating expectations and frequencies over unlabelled train+test+original data.
4. **Leak-Free Original Data Teacher**: A LightGBM model trained strictly on the disjoint original dataset outputs out-of-domain prior logits `orig_proba` and `orig_logit`.

def add_domain_features(df):
    df = df.copy()
    r = df[RATE_COLS]
    
    # Rating aggregates
    df['rate_mean'] = r.mean(axis=1)
    df['rate_std'] = r.std(axis=1)
    df['rate_sum'] = r.sum(axis=1)
    df['rate_min'] = r.min(axis=1)
    df['rate_max'] = r.max(axis=1)
    df['rate_median'] = r.median(axis=1)
    
    # Discrete behavioral counters
    df['n_zero'] = (r == 0).sum(axis=1)
    df['n_five'] = (r == 5).sum(axis=1)
    df['n_low'] = (r <= 2).sum(axis=1)
    df['n_high'] = (r >= 4).sum(axis=1)
    
    # Delay dynamics
    df['Arrival Delay in Minutes'] = df['Arrival Delay in Minutes'].fillna(df['Departure Delay in Minutes'])
    df['total_delay'] = df['Departure Delay in Minutes'] + df['Arrival Delay in Minutes']
    df['delay_diff'] = df['Departure Delay in Minutes'] - df['Arrival Delay in Minutes']  # Recovery in air
    df['has_delay'] = (df['total_delay'] > 0).astype(int)
    df['dep_delay_log'] = np.log1p(df['Departure Delay in Minutes'])
    df['arr_delay_log'] = np.log1p(df['Arrival Delay in Minutes'])
    df['delay_per_dist'] = df['total_delay'] / (df['Flight Distance'] + 1)
    
    # Composite interaction indices
    df['wifi_x_online'] = df['Inflight wifi service'] * df['Online boarding']
    df['comfort_sum'] = df['Seat comfort'] + df['Leg room service'] + df['Inflight entertainment']
    df['online_sum'] = df['Inflight wifi service'] + df['Ease of Online booking'] + df['Online boarding']
    df['flight_dist_log'] = np.log1p(df['Flight Distance'])
    
    return df

def load_original_dataset():
    if not os.path.exists(f'{ORIG}/train.csv'):
        return None
    o = pd.concat([pd.read_csv(f'{ORIG}/train.csv'), pd.read_csv(f'{ORIG}/test.csv')], ignore_index=True)
    o = o.drop(columns=[c for c in ['Unnamed: 0', 'Inflight service'] if c in o.columns])
    o['satisfaction'] = (o['satisfaction'] == 'satisfied').astype(int)
    return add_domain_features(o)

import lightgbm as lgb

def build_full_features(train_df, test_df):
    tr = add_domain_features(train_df)
    te = add_domain_features(test_df)
    orig = load_original_dataset()
    
    # Concatenate all unlabelled matrices for label-free route profiling
    if orig is not None:
        all_x = pd.concat([tr.drop(columns=['satisfaction']), te, orig.drop(columns=['satisfaction'])], ignore_index=True)
    else:
        all_x = pd.concat([tr.drop(columns=['satisfaction']), te], ignore_index=True)
        
    KEY = 'Flight Distance'
    gm = all_x.groupby(KEY)[RATE_COLS + ['Age', 'total_delay']].mean()
    gs = all_x.groupby(KEY)[RATE_COLS].std()
    cnt = all_x.groupby(KEY).size()
    total_rows = len(all_x)
    
    for df in (tr, te):
        for c in RATE_COLS:
            df[f'route_{c}_mean'] = df[KEY].map(gm[c])
            df[f'route_{c}_std'] = df[KEY].map(gs[c])
            df[f'{c}_resid'] = df[c] - df[f'route_{c}_mean']
        df['route_Age_mean'] = df[KEY].map(gm['Age'])
        df['route_delay_mean'] = df[KEY].map(gm['total_delay'])
        df['route_count'] = df[KEY].map(cnt)
        df['route_freq'] = np.log1p(1e6 * df['route_count'] / total_rows)
        
    for c in ['Class', 'Type of Travel', 'Customer Type', 'Gender']:
        sh = pd.get_dummies(all_x[[KEY, c]], columns=[c]).groupby(KEY).mean()
        for col in sh.columns:
            tr[f'rp_{col}'] = tr[KEY].map(sh[col])
            te[f'rp_{col}'] = te[KEY].map(sh[col])
            
    # Leak-free Teacher Model from Original Dataset
    if orig is not None:
        base_features = [c for c in orig.columns if c in tr.columns and pd.api.types.is_numeric_dtype(orig[c]) and c not in ('id', 'satisfaction')]
        teacher = lgb.LGBMClassifier(objective='binary', n_estimators=400, learning_rate=0.05, num_leaves=63,
                                     subsample=0.8, colsample_bytree=0.8, verbose=-1, n_jobs=-1, random_state=CFG.seed)
        teacher.fit(orig[base_features], orig['satisfaction'].values)
        for df in (tr, te):
            p = np.clip(teacher.predict_proba(df[base_features])[:, 1], 1e-6, 1 - 1e-6)
            df['orig_proba'] = p
            df['orig_logit'] = np.log(p / (1 - p))
            
        rate = orig.groupby(KEY)['satisfaction'].mean()
        gmean = orig['satisfaction'].mean()
        tr['orig_route_sat'] = tr[KEY].map(rate).fillna(gmean)
        te['orig_route_sat'] = te[KEY].map(rate).fillna(gmean)
    else:
        for df in (tr, te):
            df['orig_proba'] = 0.5
            df['orig_logit'] = 0.0
            df['orig_route_sat'] = 0.5
            
    # Conditional combinations for Target Encoding
    cond_keys = []
    for c in RATE_COLS:
        for s in ['Type of Travel', 'Customer Type', 'Class']:
            nm = f'cond_{c}__{s}'
            cond_keys.append(nm)
            tr[nm] = tr[c].astype(str) + '_' + tr[s].astype(str)
            te[nm] = te[c].astype(str) + '_' + te[s].astype(str)
            
    return tr, te, cond_keys

trf, tef, COND = build_full_features(train, test)
gc.collect()
print(f"✨ Feature matrix generated: {trf.shape[1]} columns, {len(COND)} conditional keys.")

## 4. GPU-Accelerated 5-Fold Stratified Base Models

Inside every fold, a `TargetEncoder` is fit strictly on the training partition (`cv=5`, `smooth='auto'`) and transformed onto validation and test sets to guarantee zero leakage. We train **LightGBM** (with GPU-compatible `max_bin=255`), **XGBoost** (`tree_method='hist'`), and **CatBoost** (`task_type='GPU'`).

def add_categorical_combos(df):
    df = df.copy()
    df['cmb_FD_Class'] = df['Flight Distance'].astype(str) + '_' + df['Class'].astype(str)
    df['cmb_FD_Travel'] = df['Flight Distance'].astype(str) + '_' + df['Type of Travel'].astype(str)
    df['cmb_FD_Cust'] = df['Flight Distance'].astype(str) + '_' + df['Customer Type'].astype(str)
    df['cmb_seg'] = df['Customer Type'].astype(str) + '_' + df['Type of Travel'].astype(str) + '_' + df['Class'].astype(str)
    df['FD_str'] = df['Flight Distance'].astype(str)
    df['Age_str'] = df['Age'].astype(str)
    return df

trf = add_categorical_combos(trf)
tef = add_categorical_combos(tef)

TE_COLS = ['FD_str', 'Age_str', 'cmb_FD_Class', 'cmb_FD_Travel', 'cmb_FD_Cust', 'cmb_seg'] + COND
for c in CAT_COLS:
    trf[c] = trf[c].astype('category')
    tef[c] = tef[c].astype('category')

DROP_COLS = ['id', 'satisfaction']
NUM_COLS = [c for c in trf.columns if c not in DROP_COLS and c not in TE_COLS and c not in CAT_COLS]
print(f"🔤 Numerical Features: {len(NUM_COLS)} | Target Encoding Features: {len(TE_COLS)}")

import xgboost as xgb
from catboost import CatBoostClassifier

def check_gpu():

GPU = check_gpu()

def create_fold_matrices(train_idx, val_idx):
    enc = TargetEncoder(smooth='auto', cv=5, random_state=CFG.seed, target_type='binary')
    te_names = ['te_' + c for c in TE_COLS]
    
    X_tr = pd.concat([
        trf.iloc[train_idx][NUM_COLS + CAT_COLS].reset_index(drop=True),
        pd.DataFrame(enc.fit_transform(trf.iloc[train_idx][TE_COLS], y[train_idx]), columns=te_names)
    ], axis=1)
    
    X_va = pd.concat([
        trf.iloc[val_idx][NUM_COLS + CAT_COLS].reset_index(drop=True),
        pd.DataFrame(enc.transform(trf.iloc[val_idx][TE_COLS]), columns=te_names)
    ], axis=1)
    
    X_te = pd.concat([
        tef[NUM_COLS + CAT_COLS].reset_index(drop=True),
        pd.DataFrame(enc.transform(tef[TE_COLS]), columns=te_names)
    ], axis=1)
    
    for c in CAT_COLS:
        X_tr[c] = X_tr[c].astype('category')
        X_va[c] = X_va[c].astype('category')
        X_te[c] = X_te[c].astype('category')
        
    return X_tr, X_va, X_te

def train_base_model(model_name):
    skf = StratifiedKFold(CFG.n_folds, shuffle=True, random_state=CFG.seed)
    oof = np.zeros(len(trf), dtype=np.float32)
    preds = np.zeros(len(tef), dtype=np.float32)
    last_model = None
    
    print(f"\n🚀 Training 5-Fold {model_name.upper()}...")
    for f, (tr_idx, va_idx) in enumerate(skf.split(trf, y)):
        X_tr, X_va, X_te = create_fold_matrices(tr_idx, va_idx)
        
        if model_name == 'lgb':
            params = dict(
                objective='binary', metric='auc', learning_rate=0.03, num_leaves=96,
                feature_fraction=0.6, bagging_fraction=0.85, bagging_freq=1,
                min_child_samples=80, n_estimators=6000, verbose=-1, n_jobs=-1,
                lambda_l1=3.0, lambda_l2=3.0, max_bin=255, random_state=CFG.seed + f
            )
            if GPU:
                params['device'] = 'gpu'
            try:
                m = lgb.LGBMClassifier(**params)
                m.fit(X_tr, y[tr_idx], eval_set=[(X_va, y[va_idx])], eval_metric='auc',
                      categorical_feature=CAT_COLS, callbacks=[lgb.early_stopping(150, verbose=False)])
            except Exception:
                params.pop('device', None)
                m = lgb.LGBMClassifier(**params)
                m.fit(X_tr, y[tr_idx], eval_set=[(X_va, y[va_idx])], eval_metric='auc',
                      categorical_feature=CAT_COLS, callbacks=[lgb.early_stopping(150, verbose=False)])
        elif model_name == 'xgb':
            m = xgb.XGBClassifier(
                n_estimators=6000, learning_rate=0.03, max_depth=8, subsample=0.85,
                colsample_bytree=0.6, min_child_weight=5, reg_lambda=3.0, reg_alpha=1.0,
                tree_method='hist', device='cuda' if GPU else 'cpu', eval_metric='auc',
                enable_categorical=True, early_stopping_rounds=150, n_jobs=-1, random_state=CFG.seed + f
            )
            m.fit(X_tr, y[tr_idx], eval_set=[(X_va, y[va_idx])], verbose=False)
        else:
            m = CatBoostClassifier(
                iterations=2500, learning_rate=0.05, depth=8, l2_leaf_reg=3.0,
                eval_metric='AUC', cat_features=CAT_COLS, random_seed=CFG.seed + f,
                early_stopping_rounds=150, verbose=False, task_type='GPU' if GPU else 'CPU'
            )
            m.fit(X_tr, y[tr_idx], eval_set=(X_va, y[va_idx]))
            
        val_pred = m.predict_proba(X_va)[:, 1]
        oof[va_idx] = val_pred
        preds += m.predict_proba(X_te)[:, 1] / CFG.n_folds
        fold_auc = roc_auc_score(y[va_idx], val_pred)
        print(f"  • Fold {f} ROC AUC: {fold_auc:.5f}")
        last_model = m
        del X_tr, X_va, X_te, m
        gc.collect()
        
    total_auc = roc_auc_score(y, oof)
    print(f"  ==> Overall {model_name.upper()} 5-Fold OOF AUC: {total_auc:.5f}")
    return oof, preds, last_model

own_oof, own_pred, models = {}, {}, {}
for model_type in ['lgb', 'xgb', 'cat']:
    try:
        oof_p, test_p, mdl = train_base_model(model_type)
        own_oof[model_type] = oof_p
        own_pred[model_type] = test_p
        models[model_type] = mdl
    except Exception as e:
        print(f"⚠️ Skipped {model_type}: {e}")

    # Cap to top 90 & deduplicate near-copies (r > 0.999)

    K = min(90, len(names))

    OOF, PRED = OOF[:, kp], PRED[:, kp]
    
    R = np.column_stack([rankdata(OOF[:, i]).astype('float32') for i in range(OOF.shape[1])])

    del R; gc.collect()
    
    def logit_func(a, eps=1e-6):
        a = np.clip(a, eps, 1 - eps)
        return np.log(a / (1 - a))
