def plot_kde(df1: pd.DataFrame,
             df2: pd.DataFrame,
             cols: int = 3,
             fill: bool = True,
             figsize: tuple = (15, 12),
             colors: list = ["#009E73", "#D55E00"],
             title: str = None,
             labels=("Train", "Test")) -> None:

    df1 = df1.copy()
    df2 = df2.copy()

    df1.drop([CFG.target], axis=1, inplace=True)

    numeric_columns = df1.select_dtypes(include='number').columns
    rows = np.ceil(len(numeric_columns) / cols).astype(int)

    fig, ax = plt.subplots(nrows=rows, ncols=cols, figsize=figsize)
    fig.patch.set_facecolor("#FFFCFC")

    for i, x in enumerate(numeric_columns):
        current_ax = ax[i // cols, i % cols] if rows > 1 else ax[i % cols]
        current_ax.set_facecolor("#FFFCFC")
        sns.kdeplot(x=x, color=colors[0], fill=fill, data=df1, label=labels[0], ax=current_ax, linewidth=2.5)
        sns.kdeplot(x=x, color=colors[1], fill=fill, data=df2, label=labels[1], ax=current_ax, linewidth=2.5)
        current_ax.set_xlabel(x, fontsize=14, weight="bold")
        current_ax.set_ylabel("Density", fontsize=12)
        current_ax.tick_params(labelsize=11)

        current_ax.spines["left"].set_visible(False)
        current_ax.spines["bottom"].set_visible(True)
        current_ax.spines["right"].set_visible(False)
        current_ax.spines["top"].set_visible(False)

        current_ax.spines["bottom"].set_color("#000")
        current_ax.spines["bottom"].set_linewidth(2)
        
        current_ax.xaxis.grid(False)
        current_ax.yaxis.grid(True, color="#000", linestyle="--", linewidth=1.0, alpha=0.2, zorder=0)
        current_ax.set_axisbelow(True)

    if len(numeric_columns) < (rows * cols):
        for i in range(len(numeric_columns), rows * cols):
            fig.delaxes(np.atleast_1d(ax).flatten()[i])

    legend_elements = [
        Patch(facecolor=colors[0], edgecolor='black', label=labels[0]),
        Patch(facecolor=colors[1], edgecolor='black', label=labels[1])
    ]

    fig.legend(handles=legend_elements, loc='upper center', ncol=2,
               bbox_to_anchor=(0.5, 0.97), frameon=False, fontsize=13)

    if title:
        fig.suptitle(title, fontsize=22, weight="bold", y=0.99)

    fig.subplots_adjust(left=0.07, right=0.97, top=0.90, bottom=0.07, hspace=0.5, wspace=0.3)
    plt.show()

def read_data(directory:str) -> tuple:
    """
        Read CSV files from a directory and return train, test, and submission DataFrames.
    """
    data = {}
    for filename in os.listdir(directory):
        key = re.search("(.*).csv", filename).group(1)
        data[key] = pd.read_csv(
            os.path.join(directory, filename)
        )

    train = data["train"]
    test = data["test"]
    sub = data["sample_submission"]

    return train, test, sub

<div style="background:#FFF; border:1px solid #000; border-left:5px solid #000; border-radius:14px; padding:22px 26px; margin:12px 0;">
<p>Let's try to detect covariate shift using PCA. PCA compresses the data into a lower-dimensional space, but this compression involves some information loss. This means that if we reconstruct the data from its compressed representation, the result won't exactly match the original data. The difference between the original data and the PCA reconstruction is called the reconstruction error.</p>
<p>The general idea is the following. We fit the PCA on a reference dataset R and measure the reconstruction error on data that the PCA never saw (a held-out part of R). If another dataset D has the same joint distribution as R, the PCA should reconstruct D about as well as it reconstructs the held-out R, so both errors will be similar. If the joint distribution has shifted, the directions learned from R no longer describe D well, and the reconstruction error changes.</p>
<p>One small caveat before we start, PCA only captures changes in means, variances and linear correlations, so a purely non-linear shift can go undetected.</p>
</div>

def reconstruction_error_per_row(data):
    scaled = scaler.transform(data)
    reconstructed = pca.inverse_transform(pca.transform(scaled))
    return np.sqrt(((scaled - reconstructed) ** 2).sum(axis=1))


def reconstruction_error_per_chunk(data, chunk_size):
    row_errors = reconstruction_error_per_row(data)
    shuffled_positions = np.random.default_rng(CFG.seed).permutation(len(row_errors))
    n_chunks = len(row_errors) // chunk_size

    return np.array([
        row_errors[shuffled_positions[i * chunk_size:(i + 1) * chunk_size]].mean()
        for i in range(n_chunks)
    ])

reference_std = reference_errors.std(ddof=1)

analysis_outside_bounds = (analysis_errors < lower_bound) | (analysis_errors > upper_bound)

<div style="background:#FFF; border:1px solid #000; border-left:5px solid #000; border-radius:14px; padding:22px 26px; margin:12px 0;">
<p>Let's test whether those values are Missing Completely At Random (MCAR). We have two hypotheses:</p>
<ul>
<li>Null hypothesis: the data are Missing Completely At Random (MCAR).</li>
<li>Alternative hypothesis: the data are not MCAR.</li>
</ul>
<p>Under MCAR, missingness in Arrival Delay in Minutes is independent of all other variables. A necessary consequence is that, for every other variable X, the mean of X should be the same in rows where Arrival Delay in Minutes is missing as in rows where it is present:</p>
</div>

$$E[X \mid \text{Arrival Delay is missing}] = E[X \mid \text{Arrival Delay is present}]$$




<div style="background:#FFF; border:1px solid #000; border-left:5px solid #000; border-radius:14px; padding:22px 26px; margin:12px 0;">
<p>So we just want to calculate how likely it is to observe a difference at least as large as the one in the data, given that the null hypothesis is true. If that probability p is very low, then what we observe would be very unlikely under the null hypothesis. So we reject it.</p> <p>That is what happens in this case (given an alpha of 0.001). If Departure Delay were independent of whether Arrival Delay is missing, a difference this large between the two group means would be extremely unlikely (p = 3.52e-61).</p>
</div>

# Simplified version of Little's MCAR test (chi-square on the difference between the group means). You can read more about the original here: https://journals.sagepub.com/doi/pdf/10.1177/1536867X1301300407
def little_mcar(df, col):
    missing = df[col].isna()
    X = df.drop(columns=col).astype(float)

    diff = X[missing].mean() - X[~missing].mean()
    d = diff.to_numpy()
    cov = np.cov(X.to_numpy(), rowvar=False, bias=True)

    chi2 = missing.sum() * (~missing).sum() / len(X) * d @ np.linalg.pinv(cov) @ d
    dof = np.linalg.matrix_rank(cov)

    return chi2, dof, stats.chi2.sf(chi2, dof)

alpha = 0.001
chi2, dof, p = little_mcar(train, "Arrival Delay in Minutes")
print(f"chi2 = {chi2:.1f} | dof = {dof} | p = {p:.3g} | H0 rejected: {p < alpha}")

def apply_features(df, train, test, original):
    original = original.copy()
    train = train.copy()
    df = df.copy()
    
    numeric_cols = ["Age", "Flight Distance", "Departure Delay in Minutes", "Arrival Delay in Minutes"]
    categorical_cols = ["Customer Type", "Type of Travel", "Class"]
    rating_cols = ["Inflight wifi service", "Departure/Arrival time convenient", "Ease of Online booking", "Gate location", "Food and drink", "Online boarding", "Seat comfort", "Inflight entertainment", "On-board service", "Leg room service", "Baggage handling", "Checkin service", "Cleanliness"]

    # Missing
    df["arr_missing"] = df["Arrival Delay in Minutes"].isna().astype(int)
    df["Arrival Delay in Minutes"] = df["Arrival Delay in Minutes"].fillna(df["Departure Delay in Minutes"])

    # Interactions
    df["delay_total"] = df["Departure Delay in Minutes"] + df["Arrival Delay in Minutes"]
    df["delay_diff"] = df["Arrival Delay in Minutes"] - df["Departure Delay in Minutes"]
    df["delay_recovered"] = df["Departure Delay in Minutes"] - df["Arrival Delay in Minutes"]
    df["delay_recovered_flag"] = (df["delay_recovered"] > 0).astype(int)
    df["delay_per_distance"] = df["Arrival Delay in Minutes"] / (df["Flight Distance"] + 1)
    
    # Stats
    distance_group = pd.concat([train, test]).groupby("Flight Distance")
    df["flight_distance_count"] = df["Flight Distance"].map(pd.concat([train, test])["Flight Distance"].value_counts()).fillna(0)

    services = ["Seat comfort", "Cleanliness", "Inflight entertainment", "Food and drink"]
    services_clean = df[services].replace(0, np.nan).fillna(0)
    df["cabin_score"] = services_clean.mean(axis=1)
    df["cabin_min"] = services_clean.min(axis=1)
    df["cabin_std"] = services_clean.std(axis=1)
    df["cabin_unrated_count"] = (df[services] == 0).sum(axis=1)

    digital = ["Inflight wifi service", "Ease of Online booking", "Online boarding"]
    df["digital_score"] = df[digital].mean(axis=1)
    df["digital_min"] = df[digital].min(axis=1)

    df["distance_vs_class"] = df["Flight Distance"] - df["Class"].map(train.groupby("Class")["Flight Distance"].mean())

    # Ratings
    df["rating_zeros"] = (df[rating_cols] == 0).sum(axis=1)
    df["rating_high"] = (df[rating_cols] >= 4).sum(axis=1)
    for col in ["Seat comfort", "Leg room service", "Inflight entertainment"]:
        df[f"{col}_high"] = (df[col] >= 4).astype(int)
    df["Cleanliness_high"] = (df["Cleanliness"] >= 3).astype(int)

    # Frequency encoding
    for col in numeric_cols:
        df[f"{col}_category"] = df[col].astype(int).astype(str)
        df[f"{col}_frequency"] = df[col].map(train[col].value_counts()).fillna(0)

    # Convert ratings to categorical features
    for col in rating_cols:
        df[f"{col}_category"] = df[col].astype(str)

    # Mean of each feature within the same flight-distance group
    for col in rating_cols + numeric_cols:
        df[f"{col}_flight_distance_mean"] = df["Flight Distance"].map(distance_group[col].mean())

    # Mean category code within the same flight-distance group
    for col in categorical_cols:
        codes = pd.concat([train, test])[col].astype("category").cat.codes
        df[f"{col}_flight_distance_mean"] = df["Flight Distance"].map(codes.groupby(pd.concat([train, test])["Flight Distance"]).mean())

    # Original dataset statistics
    prior = original[CFG.target].mean()
    groups = {
        "distance": ["Flight Distance"],
        "class_travel_customer": ["Class", "Type of Travel", "Customer Type"],
        "boarding_wifi": ["Online boarding", "Inflight wifi service"]
    }

    for name, cols in groups.items():
        stats = original.groupby(cols)[CFG.target].agg(["sum", "count"]).reset_index()
        stats["target_mean"] = (stats["sum"] + 20 * prior) / (stats["count"] + 20)
        df = df.merge(stats[cols + ["target_mean", "count"]], on=cols, how="left")
        df[f"original_target_mean_{name}"] = df.pop("target_mean").fillna(prior)
        df[f"original_count_{name}"] = df.pop("count").fillna(0)

    for col in numeric_cols:
        df[f"original_frequency_{col}"] = df[col].map(original[col].value_counts()).fillna(0)

    return df

def hill_climbing(X, y, model):
    feature_groups = {
        "stats": ["flight_distance_count", "cabin_score", "cabin_min", "cabin_std", "digital_score", "digital_min", "distance_vs_class"],
        "ratings": ["rating_zeros", "rating_high", "Seat comfort_high", "Leg room service_high", "Inflight entertainment_high", "Cleanliness_high"],
        "delay": ["delay_total", "delay_diff", "delay_recovered", "delay_recovered_flag", "delay_per_distance"],
        "frequency": [col for col in X if col.endswith("_frequency")],
        "category": [col for col in X if col.endswith("_category")],
        "flight_distance_mean": [col for col in X if col.endswith("_flight_distance_mean")],
        "original": [col for col in X if col.startswith("original_")]
    }

    features = [col for col in X if not any(col in group for group in feature_groups.values())]
    X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=0.2, stratify=y, random_state=CFG.seed)

    model.fit(X_train[features], y_train)
    best_score = roc_auc_score(y_val, model.predict_proba(X_val[features])[:, 1])
    print(f"Baseline: {best_score:.6f}", flush=True)

    selected = []

    while True:
        scores = {}

        for group, group_features in feature_groups.items():
            if group not in selected:
                candidate = features.copy()
                for selected_group in selected:
                    candidate += feature_groups[selected_group]
                candidate += group_features
                model.fit(X_train[candidate], y_train)
                scores[group] = roc_auc_score(y_val, model.predict_proba(X_val[candidate])[:, 1])

        group = max(scores, key=scores.get)

        if scores[group] <= best_score:
            break

        selected.append(group)
        best_score = scores[group]
        print(f"{group}: {best_score:.6f}", flush=True)

    for group in selected:
        features += feature_groups[group]

    return features

def cross_validation(model, X, y, test, n_splits=CFG.splits):
    cv = StratifiedKFold(n_splits, shuffle=True, random_state=CFG.seed)
    categorical = X.select_dtypes(exclude="number").columns.tolist()
    
    oof_train = np.zeros(len(X))
    oof_test = np.zeros(len(test))
    oof_train_logits = np.zeros(len(X))
    oof_test_logits = np.zeros(len(test))
    scores = []

    for i, (train_idx, test_idx) in enumerate(cv.split(X, y)):
        X_train, y_train = X.iloc[train_idx], y.iloc[train_idx]
        X_test, y_test = X.iloc[test_idx], y.iloc[test_idx]

        X_fit, X_stop, y_fit, y_stop = train_test_split(
            X_train, y_train, test_size=0.1, stratify=y_train, random_state=CFG.seed
        )

        clf = clone(model).fit(
            X_fit,
            y_fit,
            eval_set=(X_stop, y_stop),
            cat_features=categorical,
            early_stopping_rounds=300,
        )

        y_pred = clf.predict_proba(X_test)[:, 1]
        
        oof_train[test_idx] = y_pred
        oof_train_logits[test_idx] = clf.predict(X_test, prediction_type="RawFormulaVal")
        oof_test += clf.predict_proba(test)[:,1]/n_splits
        oof_test_logits += clf.predict(test, prediction_type="RawFormulaVal")/n_splits

        rauc = roc_auc_score(y_test, y_pred)
        scores.append(rauc)
        
        print(f"Fold {i + 1}: {scores[-1]:.4f}")
        
    print(f"Overall: {np.mean(scores):.4f} ± {np.std(scores):.4f}")

    return {
        "train": oof_train,
        "test": oof_test,
        "train_logits": oof_train_logits,
        "test_logits": oof_test_logits,
        "scores": scores,
        "mean_score": np.mean(scores),
    }
